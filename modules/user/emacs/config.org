  (evil-define-operator nj/align-equals (beg end)
    "Indent and align equal sign, this aligns every continous text that has `=`, maybe this should only
    work for the operators like `+` or `*` but I'm not sure."
    (interactive "<r>")
    (evil-indent beg end)
    (align-regexp beg end "\\(\\s-*\\)\\S-*=" 1 1 t))

  (evil-define-operator nj/align-char (beg end char)
    "Indent and align a single character."
    (interactive "<r>")
    (evil-indent beg end)
    (align-regexp beg end
                  (format "\\(\\s-*\\)%s" (regexp-quote (char-to-string (read-char-exclusive char))))
                  1 1 t))

#+begin_src emacs-lisp
  (defun nj/imenu--flatten (Index)
    (let (Items)
      (dolist (Entry Index Items)
        (cond
         ;; NOTE: Skip invalid entries
         ((null (car-safe Entry)))
         ((string= (car Entry) "*Rescan*"))
         ((imenu--subalist-p Entry)
          (setq Items (nconc Items (nj/imenu--flatten (cdr Entry)))))
      (t
       (push Entry Items))))))

  (defun nj/imenu ()
    (interactive)
    (let* ((Project (project-current))
           (Root    (if Project
                        (project-root Project)
                      default-directory))
           (Files   (if Project
                        (project-files Project)
                      (directory-files-recursively Root ".")))
           (Items ()))
      (dolist (File Files)
        (when (and (file-regular-p File)
                   (not (backup-file-name-p File)))
          (let ((Buf (find-file-noselect File t)))
            (with-current-buffer Buf
              (ignore-errors
                (let ((Index (nj/imenu--flatten (imenu--make-index-alist t))))
                  (dolist (Entry Index)
                    (unless (string= (car Entry) "*Rescan*")
                      (let* ((Func   (car Entry))
                             (Marker (cdr Entry))
                             (Line   (when (markerp Marker)
                                       (save-excursion
                                         (goto-char Marker)
                                         (string-trim
                                          (buffer-substring-no-properties
                                           (line-beginning-position)
                                           (line-end-position))))))
                             (Label (if Line
                                        (format "%s %s" Func
                                                (propertize Line 'face '(:foreground "Red"))))))
                        (push (cons Label Marker) Items))))))))))
      (if (null Items)
          (message "No imenu entries found.")
        (let* ((Prompt (completing-read "Jump to: " Items nil t))
               (Entry  (assoc Prompt Items)))
          (when Entry
            (let ((Marker (cdr Entry)))
              (if (markerp Marker)
                  (progn
                    (switch-to-buffer (marker-buffer Marker))
                    (goto-char Marker)) ;; NOTE: Recenter?
                (imenu Entry))))))))

  (defun nj/open-remote ()
    "Open remote repository in the browser."
    (interactive)
    (if (project-current)
        (let ((url (string-trim (shell-command-to-string "git remote get-url origin"))))
          (if (string-match-p "github\\.com" url)
              (browse-url (string-replace "git@github.com:" "https://github.com/" url))
            (message "ERROR: The repository: %s is not hosted on GitHub." url)))
      (message "ERROR: Not a git repository.")))

  (defun nj/markdown-insert-link-from-clipboard ()
    "Insert a markdown link with the contents of the system clipboard."
    (interactive)
    (let* ((title (read-string "Title: "))
           (clip  (string-trim (evil-get-register ?+))))
      (insert (format "[%s](%s)" title clip))))

  (defmacro nj/project-or-default-directory (&rest Body)
    `(let ((default-directory
            (or (when-let ((P (project-current)))
                  (project-root P))

       ,@Body))

  (defun nj-find-corresponding-file ()
    (interactive)
    (if (member (file-name-extension buffer-file-name)

        (if-let ((File

                    (goto-char (point-min))

#+begin_src emacs-lisp
  (defun nj/project-init ()
    "Create a git repository with direnv and nix setup"
    (interactive)
    ;; Build packages list from input
    (let* ((dir   (read-directory-name "Create repository in: "))
           (default-directory (file-name-as-directory dir))
           (env-file ".envrc")
           (flake-file "flake.nix")
           (input (read-string "Packages: "))
           (pkgs  (split-string input "[[:space:]]+" t))
           (pkgs-list
            (concat
             "                      "
             (mapconcat #'identity pkgs "\n                      ")
             "\n")))

      (make-directory default-directory t)

      (with-temp-file env-file
        (insert "use flake"))
      (envrc-allow)

      (with-temp-file flake-file
        (insert
         "{\n"
         "    description = \"A very basic flake\";\n"
         "\n"
         "    inputs = {\n"
         "        nixpkgs.url = \"nixpkgs/nixos-unstable\";\n"
         "    };\n"
         "\n"
         "    outputs =\n"
         "        { self, nixpkgs }:\n"
         "        let\n"
         "            system = \"x86_64-linux\";\n"
         "            pkgs = import nixpkgs { inherit system; };\n"
         "        in\n"
         "          {\n"
         "              devShells.${system}.default = pkgs.mkShell {\n"
         "                  buildInputs = with pkgs; [\n"
         pkgs-list
         "                  ];\n"
         "              };\n"
         "          };\n"
         "}"))
      (envrc-reload)
      (unless (project-current)
        (call-process "git" nil nil nil "init"))
      (call-process "git" nil nil nil "add" env-file flake-file "flake.lock")))
#+end_src
